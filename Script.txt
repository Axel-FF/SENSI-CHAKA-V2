local function Cleanup()
	S.ESP, S.Aimbot, S.Rage, S.Legit, S.AimKill, S.Chams, S.Rainbow = false, false, false, false, false, false, false
	if MainConnection then MainConnection:Disconnect(); MainConnection = nil end
	if FOVCircle then pcall(function() FOVCircle.Visible = false; FOVCircle:Remove() end); FOVCircle = nil end
	for _, plr in pairs(ESPObjects) do
		if ESPObjects[plr] then for _, o in pairs(ESPObjects[plr]) do pcall(function() o:Remove() end) end end
	end
	for _, hl in pairs(ChamsObjects) do pcall(function() hl:Destroy() end) end
	table.clear(ESPObjects); table.clear(ChamsObjects)
end
CloseBtn.MouseButton1Click:Connect(function() Cleanup(); gui:Destroy() end)

-- Funciones principales
local function IsEnemy(plr)
	if not S.TeamCheck then return true end
	if not LP:FindFirstChild("Team") or not plr:FindFirstChild("Team") then return true end
	return LP.Team ~= plr.Team
end

local function IsVisible(part)
	if not S.WallCheck then return true end
	local origin = Camera.CFrame.Position
	local rayParams = RaycastParams.new()
	rayParams.FilterDescendantsInstances = {LP.Character or {}, part.Parent}
	rayParams.FilterType = Enum.RaycastFilterType.Exclude
	local dir = part.Position - origin
	return Workspace:Raycast(origin, dir.Unit * dir.Magnitude, rayParams) == nil
end

local function GetBodyPart(char, name)
	return char:FindFirstChild(name) or char:FindFirstChild(name.."Part") or char:FindFirstChild("Upper"..name) or char:FindFirstChild("Lower"..name)
end

local function GetTargetPart(char)
	if S.AimPart == "Torso" then
		return GetBodyPart(char, "Torso") or char:FindFirstChild("HumanoidRootPart")
	elseif S.AimPart == "Root" then
		return char:FindFirstChild("HumanoidRootPart") or GetBodyPart(char, "Torso")
	else
		return GetBodyPart(char, "Head") or char:FindFirstChild("HumanoidRootPart")
	end
end

local function GetClosest()
	local closest, best = nil, S.FOV
	local center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
	for _, plr in pairs(Players:GetPlayers()) do
		if plr ~= LP and plr.Character and IsEnemy(plr) then
			local hum = plr.Character:FindFirstChildOfClass("Humanoid")
			if hum and hum.Health > 0 then
				local part = GetTargetPart(plr.Character)
				if part then
					local pos, onScreen = Camera:WorldToViewportPoint(part.Position)
					if onScreen then
						local dist = (Vector2.new(pos.X, pos.Y) - center).Magnitude
						if dist < best and IsVisible(part) then
							best = dist; closest = part
						end
					end
				end
			end
		end
	end
	return closest
end

-- Chams
local function UpdateChams(plr, char, color)
	if not S.Chams then return end
	local hl = ChamsObjects[plr] or Instance.new("Highlight")
	hl.Name = "SensiChamsV2"
	hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
	hl.FillTransparency = 0.45
	hl.OutlineTransparency = 0.1
	hl.Adornee = char
	hl.Parent = char
	hl.FillColor = color
	hl.OutlineColor = color
	ChamsObjects[plr] = hl
end

local function ClearPlayerData(plr)
	if ESPObjects[plr] then
		for _, o in pairs(ESPObjects[plr]) do pcall(function() o:Remove() end) end
		ESPObjects[plr] = nil
	end
	if ChamsObjects[plr] then pcall(function() ChamsObjects[plr]:Destroy() end); ChamsObjects[plr] = nil end
end

local function CreatePlayerESP(plr)
	if ESPObjects[plr] or not HasDrawing then return end
	local t = {}
	t.box = Drawing.new("Square"); t.box.Thickness = 1.5; t.box.Visible = false; t.box.Filled = false
	t.tracer = Drawing.new("Line"); t.tracer.Thickness = 1.2; t.tracer.Visible = false
	t.name = Drawing.new("Text"); t.name.Size = 14; t.name.Center = true; t.name.Outline = true; t.name.Visible = false
	t.dist = Drawing.new("Text"); t.dist.Size = 13; t.dist.Center = true; t.dist.Outline = true; t.dist.Visible = false
	t.hp = Drawing.new("Line"); t.hp.Thickness = 3; t.hp.Visible = false
	ESPObjects[plr] = t
end

-- Bucle principal
MainConnection = RunService.RenderStepped:Connect(function()
	Camera = Workspace.CurrentCamera
	local currentColor = S.Rainbow and Color3.fromHSV((tick() * 0.4) % 1, 1, 1) or S.ESPColor
	
	if HasDrawing then UpdateFOV() end
	
	if S.Aimbot or S.Legit or S.Rage or S.AimKill then
		local target = GetClosest()
		if target then
			if S.Rage or S.AimKill then
				Camera.CFrame = CFrame.new(Camera.CFrame.Position, target.Position)
			else
				Camera.CFrame = Camera.CFrame:Lerp(CFrame.new(Camera.CFrame.Position, target.Position), 0.15)
			end
		end
	end
	
	for _, plr in pairs(Players:GetPlayers()) do
		if plr ~= LP and plr.Character and IsEnemy(plr) then
			local hum = plr.Character:FindFirstChildOfClass("Humanoid")
			if hum and hum.Health > 0 then
				UpdateChams(plr, plr.Character, currentColor)
			else
				ClearPlayerData(plr)
			end
		elseif plr ~= LP then
			ClearPlayerData(plr)
		end
	end
	
	if not S.ESP or not HasDrawing then return end
	for _, plr in pairs(Players:GetPlayers()) do
		if plr ~= LP and plr.Character and IsEnemy(plr) then
			local hum = plr.Character:FindFirstChildOfClass("Humanoid")
			local root = plr.Character:FindFirstChild("HumanoidRootPart")
			local head = GetBodyPart(plr.Character, "Head")
			if hum and hum.Health > 0 and root and head then
				CreatePlayerESP(plr)
				local o = ESPObjects[plr]
				if o then
					local pos, onScreen = Camera:WorldToViewportPoint(root.Position)
					local headPos = Camera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0))
					local footPos = Camera:WorldToViewportPoint(root.Position - Vector3.new(0, 2.8, 0))
					if onScreen then
						local h = math.abs(headPos.Y - footPos.Y)
						local w = h * 0.55
						o.box.Size = Vector2.new(w, h)
						o.box.Position = Vector2.new(pos.X - w/2, pos.Y - h/2)
						o.box.Color = currentColor
						o.box.Visible = S.Box
						
						o.tracer.From = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y)
						o.tracer.To = Vector2.new(pos.X, pos.Y)
						o.tracer.Color = currentColor
						o.tracer.Visible = S.Tracer
						
						o.name.Text = plr.Name
						o.name.Position = Vector2.new(pos.X, headPos.Y - 18)
						o.name.Color = currentColor
						o.name.Visible = S.Name
						
						local dist = math.floor((root.Position - Camera.CFrame.Position).Magnitude)
						o.dist.Text = dist .. "m"
						o.dist.Position = Vector2.new(pos.X, footPos.Y + 4)
						o.dist.Color = currentColor
						o.dist.Visible = S.Dist
						
						local pct = math.clamp(hum.Health / math.max(hum.MaxHealth, 1), 0, 1)
						o.hp.From = Vector2.new(pos.X - w/2 - 6, pos.Y + h/2)
						o.hp.To = Vector2.new(pos.X - w/2 - 6, pos.Y + h/2 - h * pct)
						o.hp.Color = Color3.fromRGB(255 * (1 - pct), 255 * pct, 40)
						o.hp.Visible = S.Health
					else
						o.box.Visible = false; o.tracer.Visible = false; o.name.Visible = false
						o.dist.Visible = false; o.hp.Visible = false
					end
				end
			else
				ClearPlayerData(plr)
			end
		else
			ClearPlayerData(plr)
		end
	end
end)
